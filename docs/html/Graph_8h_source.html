257// Remove all outgoing edges to a vertex with given destination content.
258template <class T>
259bool Vertex<T>::removeEdge(T in)
260{
261 bool removedEdge = false;
262 auto it = adj.begin();
263 while (it != adj.end())
264 {
265 Edge<T> *edge = *it;
266 Vertex<T> *dest = edge->getDest();
267 if (dest->getInfo() == in)
268 {
269 it = adj.erase(it);
270 deleteEdge(edge);
271 removedEdge = true; // allows for multiple edges to connect the same pair of vertices (multigraph)
272 }
273 else
274 {
275 it++;
276 }
277 }
278 return removedEdge;
279}

281// Remove all outgoing edges from this vertex.

283void Vertex<T>::removeOutgoingEdges()
284{
285 auto it = adj.begin();
286 while (it != adj.end())
287 {
288 Edge<T> *edge = *it;
289 it = adj.erase(it);
290 deleteEdge(edge);
291 }
292}

402template <class T>
403void Vertex<T>::deleteEdge(Edge<T> *edge)
404{
405 Vertex<T> *dest = edge->getDest();
406 // Remove the corresponding edge from the incoming list
407 auto it = dest->incoming.begin();
408 while (it != dest->incoming.end())
409 {
410 if ((*it)->getOrig()->getInfo() == info)
411 {
412 it = dest->incoming.erase(it);
413 }
414 else
415 {
416 it++;
417 }
418 }
419 delete edge;
420}

525// Remove a vertex and all its incident edges.
526template <class T>
527bool Graph<T>::removeVertex(const T &in)
528{
529 for (auto it = vertexSet.begin(); it != vertexSet.end(); it++)
530 {
531 if ((*it)->getInfo() == in)
532 {
533 auto v = *it;
534 v->removeOutgoingEdges();
535 for (auto u : vertexSet)
536 {
537 u->removeEdge(v->getInfo());
538 }
539 vertexSet.erase(it);
540 delete v;
541 return true;
542 }
543 }
544 return false;
545}

548template <class T>
549bool Graph<T>::addEdge(const T &sourc, const T &dest, double w)
550{

553 if (v1 == nullptr || v2 == nullptr)

556 return true;

571template <class T>
572bool Graph<T>::addBidirectionalEdge(const T &sourc, const T &dest, double w)
573{
574 auto v1 = findVertex(sourc);
575 auto v2 = findVertex(dest);
576 if (v1 == nullptr || v2 == nullptr)
577 return false;
578 auto e1 = v1->addEdge(v2, w);
579 auto e2 = v2->addEdge(v1, w);
580 e1->setReverse(e2);
581 e2->setReverse(e1);
582 return true;
583}

615template <class T>
616Graph<T> &Graph<T>::operator=(Graph<T> &&other) noexcept
617{
618 if (this != &other)
619 {
620 deleteMatrix(distMatrix, vertexSet.size());
621 deleteMatrix(pathMatrix, vertexSet.size());
622 for (auto v : vertexSet)
623 delete v;
624
625 vertexSet = std::move(other.vertexSet);
626 distMatrix = other.distMatrix;
627 pathMatrix = other.pathMatrix;
628
629 other.vertexSet.clear();
630 other.distMatrix = nullptr;
631 other.pathMatrix = nullptr;
632 }
633 return *this;
634}
