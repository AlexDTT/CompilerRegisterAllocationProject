534void RegisterAllocApp::doLoadInputPairFromFolder()
535{
536 std::string dirPath = "inputs";
537 if (!fs::exists(dirPath))
538 dirPath = "tests/input";
539
540 std::vector<std::string> files;
541 if (fs::exists(dirPath))
542 {
543 for (const auto &entry : fs::recursive_directory_iterator(dirPath))
544 {
545 if (!entry.is_regular_file())
546 continue;
547 auto ext = entry.path().extension().string();
548 if (ext == ".txt" || ext == ".ranges" || ext == ".cfg" || ext == ".conf")
549 {
550 files.push_back(entry.path().string());
551 }
552 }
553 }
554
555 if (files.empty())
556 {
557 std::cout << " No candidate input files found in '" << dirPath << "'.\n";
558 return;
559 }
560
561 std::sort(files.begin(), files.end());
562
563 std::vector<int> cfgIdx;
564 std::vector<int> rngIdx;
565
566 std::cout << " Files in '" << dirPath << "':\n";
567 for (size_t i = 0; i < files.size(); ++i)
568 {
569 bool isCfg = looksLikeConfigPath(files[i]);
570 if (isCfg)
571 cfgIdx.push_back((int)i + 1);
572 else
573 rngIdx.push_back((int)i + 1);
574 std::cout << " [" << i + 1 << "] " << files[i]
575 << (isCfg ? " (config?)" : " (ranges?)") << "\n";
576 }
577 std::cout << " [0] Cancel\n";
578
579 int defaultRange = rngIdx.empty() ? 0 : rngIdx.front();
580 int defaultCfg = cfgIdx.empty() ? 0 : cfgIdx.front();
581
582 if (defaultRange > 0 && defaultCfg > 0)
583 {
584 std::cout << " Suggested pair:\n"
585 << " ranges -> [" << defaultRange << "] " << files[defaultRange - 1] << "\n"
586 << " config -> [" << defaultCfg << "] " << files[defaultCfg - 1] << "\n";
587 }
588
589 int rangesChoice = readInt("Select ranges file index: ");
590 if (rangesChoice <= 0 || rangesChoice > (int)files.size())
591 return;
592
593 int configChoice = readInt("Select config file index: ");
594 if (configChoice <= 0 || configChoice > (int)files.size())
595 return;
596
597 bool okRanges = loadRangesFile(files[rangesChoice - 1]);
598 bool okConfig = loadConfigFile(files[configChoice - 1]);
599
600 if (okRanges && okConfig)
601 {
602 std::cout << " Both files loaded successfully.\n";
603 buildGraphFromData();
604 }
605}
