22 struct UnionFind
23 {
24 std::vector<int> parent, rank_;
25
26 explicit UnionFind(int n) : parent(n), rank_(n, 0)
27 {
28 for (int i = 0; i < n; ++i)
29 parent[i] = i;
30 }
31
32 int find(int x)
33 {
34 if (parent[x] != x)
35 parent[x] = find(parent[x]);
36 return parent[x];
37 }
38
39 void unite(int a, int b)
40 {
41 a = find(a);
42 b = find(b);
43 if (a == b)
44 return;
45 if (rank_[a] < rank_[b])
46 std::swap(a, b);
47 parent[b] = a;
48 if (rank_[a] == rank_[b])
49 ++rank_[a];
50 }
51 };

54 bool rangesOverlap(const LiveRange &a, const LiveRange &b)
55 {
56 std::set<int> linesA;
57 for (const auto &p : a.points)
58 linesA.insert(p.line);
59 for (const auto &p : b.points)
60 if (linesA.count(p.line))
61 return true;
62 return false;
63 }

65 int firstLineOfRange(const LiveRange &range)
66 {
67 int best = std::numeric_limits<int>::max();
68 for (const auto &point : range.points)
69 best = std::min(best, point.line);

94 std::string formatWebPointsForLabel(const Web &web)
95 {
96 std::map<int, std::set<char>> markersByLine;
97 for (const auto &range : web.ranges)
98 for (const auto &point : range.points)
99 markersByLine[point.line].insert(point.marker);
100
101 std::ostringstream oss;
102 bool first = true;
103 for (const auto &[line, markers] : markersByLine)
104 {
105 if (!first)
106 oss << ",";
107
108 char marker = '\0';
109 const bool hasPlus = markers.count('+') > 0;
110 const bool hasMinus = markers.count('-') > 0;
111 if (hasPlus && !hasMinus)
112 marker = '+';
113 else if (hasMinus && !hasPlus)
114 marker = '-';
115
116 oss << line;
117 if (marker != '\0')
118 oss << marker;
119 first = false;
120 }
121 return oss.str();
122 }

129std::vector<Web> InterferenceGraph::buildWebs(
130 const std::map<std::string, std::vector<LiveRange>> &rawRanges)
131{
132 std::vector<Web> webs;
133 int nextId = 0;
134
135 for (const auto &[varName, ranges] : rawRanges)
136 {
137 const auto &varRanges = ranges;
138 int n = static_cast<int>(varRanges.size());
139 UnionFind uf(n);
140
141 // Merge ranges that share at least one line number.
142 for (int i = 0; i < n; ++i)
143 for (int j = i + 1; j < n; ++j)
144 if (rangesOverlap(varRanges[i], varRanges[j]))
145 uf.unite(i, j);
146
147 // Group ranges by their root component.
148 std::map<int, std::vector<int>> components;
149 for (int i = 0; i < n; ++i)
150 components[uf.find(i)].push_back(i);
151
152 for (auto &[root, indices] : components)
153 {
154 Web web;
155 web.id = nextId++;
156 web.variable = varName;
157
158 std::sort(indices.begin(), indices.end(),
159 [&](int lhs, int rhs)
160 {
161 return firstLineOfRange(varRanges[lhs]) < firstLineOfRange(varRanges[rhs]);
162 });
163
164 for (int idx : indices)
165 web.ranges.push_back(varRanges[idx]);
166 webs.push_back(std::move(web));
167 }
168 }
169
170 std::sort(webs.begin(), webs.end(),
171 [](const Web &lhs, const Web &rhs)
172 {
173 if (lhs.variable != rhs.variable)
174 return lhs.variable < rhs.variable;
175 if (firstLineOfWeb(lhs) != firstLineOfWeb(rhs))
176 return firstLineOfWeb(lhs) < firstLineOfWeb(rhs);
177 return lhs.id < rhs.id;
178 });
179
180 for (size_t i = 0; i < webs.size(); ++i)
181 webs[i].id = static_cast<int>(i);
182
183 return webs;
184}

189bool InterferenceGraph::interferes(const Web &a, const Web &b)
190{
191 // Build a map: line -> set of markers for web a, and web b.
192 std::unordered_map<int, std::set<char>> markersA, markersB;
193
194 for (const auto &r : a.ranges)
195 for (const auto &p : r.points)
196 markersA[p.line].insert(p.marker);
197
198 for (const auto &r : b.ranges)
199 for (const auto &p : r.points)
200 markersB[p.line].insert(p.marker);
201
202 // Check every line in web a against web b.
203 for (const auto &[line, mA] : markersA)
204 {
205 auto it = markersB.find(line);
206 if (it == markersB.end())
207 continue; // not in b at all
208
209 const auto &mB = it->second;
210
211 // Non-interference rule: if a ends here (only '-') and b starts here (only '+'),
212 // or vice-versa, they do not interfere at this line.
213 bool aOnlyEnd = (mA.size() == 1 && mA.count('-'));
214 bool aOnlyStart = (mA.size() == 1 && mA.count('+'));
215 bool bOnlyEnd = (mB.size() == 1 && mB.count('-'));
216 bool bOnlyStart = (mB.size() == 1 && mB.count('+'));
217
218 if ((aOnlyEnd && bOnlyStart) || (aOnlyStart && bOnlyEnd))
219 {
220 // Definition meets last-use at the same instruction: no interference.
221 continue;
222 }
223
224 // Any other overlap is an interference.
225 return true;
226 }
227 return false;
228}

233Graph<int> InterferenceGraph::buildGraph(const std::vector<Web> &webs)
234{
235 Graph<int> graph;
236
237 // Add one vertex per web.
238 for (const auto &web : webs)
239 graph.addVertex(web.id);
240
241 // Add undirected edges between interfering webs.
242 for (size_t i = 0; i < webs.size(); ++i)
243 {
244 for (size_t j = i + 1; j < webs.size(); ++j)
245 {
246 if (interferes(webs[i], webs[j]))
247 {
248 graph.addBidirectionalEdge(webs[i].id, webs[j].id, 1.0);
249 }
250 }
251 }
252
253 return graph;
254}
