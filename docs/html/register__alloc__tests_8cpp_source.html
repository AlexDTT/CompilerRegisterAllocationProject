97static bool colorsRespectInterference(const Graph<int> &graph,
98 const std::map<int, int> &webToRegister)
99{
100 for (const auto &[id, color] : webToRegister)
101 {
102 if (color < 0)
103 continue;
104
105 Vertex<int> *vertex = graph.findVertex(id);
106 EXPECT_TRUE(vertex != nullptr);
107 if (!vertex)
108 return false;
109
110 for (const auto *edge : vertex->getAdj())
111 {
112 int neighbor = edge->getDest()->getInfo();
113 auto it = webToRegister.find(neighbor);
114 if (it != webToRegister.end() && it->second >= 0 && it->second == color)
115 return false;
116 }
117 }
118 return true;
119}
