59 std::string formatWebLinesForLabel(const Web &web)
60 {
61 std::map<int, std::set<char>> markersByLine;
62 for (const auto &range : web.ranges)
63 {
64 for (const auto &point : range.points)
65 markersByLine[point.line].insert(point.marker);
66 }
67
68 std::ostringstream oss;
69 bool first = true;
70 for (const auto &[line, markers] : markersByLine)
71 {
72 if (!first)
73 oss << ",";
74
75 char marker = '\0';
76 const bool hasPlus = markers.count('+') > 0;
77 const bool hasMinus = markers.count('-') > 0;
78 if (hasPlus && !hasMinus)
79 marker = '+';
80 else if (hasMinus && !hasPlus)
81 marker = '-';
82
83 oss << line;
84 if (marker != '\0')
85 oss << marker;
86 first = false;
87 }
88 return oss.str();
89 }

105AllocationResult AllocationLogic::runAllocation(std::vector<Web> &webs,
106 Graph<int> &graph,
107 const Parameters &params)
108{
109 AllocationResult result;
110
111 try
112 {
113 switch (params.algorithm)
114 {
115 case AlgorithmType::Basic:
116 result = GraphColoring::basicColoring(graph, webs, params.numRegisters);
117 break;
118
119 case AlgorithmType::Spilling:
120 result = GraphColoring::spillingColoring(
121 graph, webs, params.numRegisters, params.algorithmParam);
122 break;
123
124 case AlgorithmType::Splitting:
125 result = GraphColoring::splittingColoring(
126 graph, webs, params.numRegisters, params.algorithmParam);
127 break;
128
129 case AlgorithmType::Free:
130 result = GraphColoring::freeColoring(graph, webs, params.numRegisters);
131 break;
132 }
133 }
134 catch (const std::runtime_error &e)
135 {
136 std::cerr << "Algorithm error: " << e.what() << "\n";
137 // Return an "all spilled" result so the rest of the pipeline can still run.
138 for (const auto &web : webs)
139 result.webToRegister[web.id] = -1;
140 result.feasible = false;
141 result.registersUsed = 0;
142 }
143
144 if (!result.feasible)
145 {
146 std::cerr << "Warning: register allocation was not feasible with "
147 << params.numRegisters << " register(s). All webs will be assigned to memory.\n";
148
149 // Infeasible basic allocations are represented as an all-memory result.
150 result.webToRegister.clear();
151 for (const auto &web : webs)
152 result.webToRegister[web.id] = -1;
153 result.registersUsed = 0;
154 result.spilledWebs = static_cast<int>(webs.size());
155 }
156
157 // Write output file.
158 if (!params.outputFile.empty())
159 {
160 OutputWriter::write(params.outputFile, webs, result.webToRegister,
161 buildMetadataLines(result, params));
162 std::cout << "Allocation written to '" << params.outputFile << "'.\n";
163 }
164
165 return result;
166}

171void AllocationLogic::printResult(const std::vector<Web> &webs,
172 const AllocationResult &result)
173{
174 std::cout << "\n--- Allocation Result ---\n";
175 std::cout << "Feasible : " << (result.feasible ? "Yes" : "No") << "\n";
176 std::cout << "Registers used: " << result.registersUsed << "\n\n";
177 std::cout << "Spilled webs : " << result.spilledWebs << "\n\n";
178
179 for (const auto &web : webs)
180 {
181 auto it = result.webToRegister.find(web.id);
182 int reg = (it != result.webToRegister.end()) ? it->second : -1;
183
184 std::string regStr = (reg >= 0) ? ("r" + std::to_string(reg)) : "M (memory)";
185 std::cout << " web" << web.id << " [" << web.variable << "] -> " << regStr << "\n";
186 for (const auto &range : web.ranges)
187 {
188 std::cout << " range: ";
189 for (size_t i = 0; i < range.points.size(); ++i)
190 {
191 if (i > 0)
192 std::cout << ',';
193 std::cout << range.points[i].line;
194 if (range.points[i].marker != '\0')
195 std::cout << range.points[i].marker;
196 }
197 std::cout << "\n";
198 }
199 }
200 std::cout << "-------------------------\n";
201}

327int AllocationLogic::estimateChromatic(const Graph<int> &graph,
328 const std::vector<Web> &webs)
329{
330 // Greedy clique lower bound: start with the highest-degree vertex and greedily
331 // extend the clique by adding neighbors that are adjacent to all current members.
332 if (webs.empty())
333 return 0;
334
335 // Find the web with the highest degree as seed.
336 int seedId = webs.front().id;
337 int maxDeg = -1;
338 for (const auto &web : webs)
339 {
340 Vertex<int> *v = graph.findVertex(web.id);
341 if (!v)
342 continue;
343 int deg = static_cast<int>(v->getAdj().size());
344 if (deg > maxDeg)
345 {
346 maxDeg = deg;
347 seedId = web.id;
348 }
349 }
350
351 std::vector<int> clique = {seedId};
352
353 // Candidate set: neighbors of the seed.
354 Vertex<int> *sv = graph.findVertex(seedId);
355 if (!sv)
356 return 1;
357
358 std::vector<int> candidates;
359 for (const auto *e : sv->getAdj())
360 candidates.push_back(e->getDest()->getInfo());
361
362 for (int c : candidates)
363 {
364 // Check that c is adjacent to all current clique members.
365 Vertex<int> *cv = graph.findVertex(c);
366 if (!cv)
367 continue;
368
369 bool adjToAll = true;
370 for (int m : clique)
371 {
372 bool found = false;
373 for (const auto *e : cv->getAdj())
374 {
375 if (e->getDest()->getInfo() == m)
376 {
377 found = true;
378 break;
379 }
380 }
381 if (!found)
382 {
383 adjToAll = false;
384 break;
385 }
386 }
387 if (adjToAll)
388 clique.push_back(c);
389 }
390
391 return static_cast<int>(clique.size());
392}
