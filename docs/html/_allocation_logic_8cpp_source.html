16AllocationResult AllocationLogic::runAllocation(std::vector<Web> &webs,
17 Graph<int> &graph,
18 const Parameters &params)
19{
20 AllocationResult result;
21
22 try
23 {
24 switch (params.algorithm)
25 {
26 case AlgorithmType::Basic:
27 result = GraphColoring::basicColoring(graph, webs, params.numRegisters);
28 break;
29
30 case AlgorithmType::Spilling:
31 result = GraphColoring::spillingColoring(
32 graph, webs, params.numRegisters, params.algorithmParam);
33 break;
34
35 case AlgorithmType::Splitting:
36 result = GraphColoring::splittingColoring(
37 graph, webs, params.numRegisters, params.algorithmParam);
38 break;
39
40 case AlgorithmType::Free:
41 result = GraphColoring::freeColoring(graph, webs, params.numRegisters);
42 break;
43 }
44 }
45 catch (const std::runtime_error &e)
46 {
47 std::cerr << "Algorithm error: " << e.what() << "\n";
48 // Return an "all spilled" result so the rest of the pipeline can still run.
49 for (const auto &web : webs)
50 result.webToRegister[web.id] = -1;
51 result.feasible = false;
52 result.registersUsed = 0;
53 }
54
55 if (!result.feasible)
56 {
57 std::cerr << "Warning: register allocation was not feasible with "
58 << params.numRegisters << " register(s). All webs will be assigned to memory.\n";
59
60 // Specification compliance: infeasible allocation must output registers: 0
61 // and assign all webs to memory.
62 result.webToRegister.clear();
63 for (const auto &web : webs)
64 result.webToRegister[web.id] = -1;
65 result.registersUsed = 0;
66 result.spilledWebs = static_cast<int>(webs.size());
67 }
68
69 // Write output file.
70 if (!params.outputFile.empty())
71 {
72 OutputWriter::write(params.outputFile, webs, result.webToRegister);
73 std::cout << "Allocation written to '" << params.outputFile << "'.\n";
74 }
75
76 return result;
77}

95 std::string regStr = (reg >= 0) ? ("r" + std::to_string(reg)) : "M (memory)";

117int AllocationLogic::estimateChromatic(const Graph<int> &graph,
118 const std::vector<Web> &webs)
119{
120 // Greedy clique lower bound: start with the highest-degree vertex and greedily
121 // extend the clique by adding neighbors that are adjacent to all current members.
122 if (webs.empty())
123 return 0;
124
125 // Find the web with the highest degree as seed.
126 int seedId = webs.front().id;
127 int maxDeg = -1;
128 for (const auto &web : webs)
129 {
130 Vertex<int> *v = graph.findVertex(web.id);
131 if (!v)
132 continue;
133 int deg = static_cast<int>(v->getAdj().size());
134 if (deg > maxDeg)
135 {
136 maxDeg = deg;
137 seedId = web.id;
138 }
139 }
140
141 std::vector<int> clique = {seedId};
142
143 // Candidate set: neighbors of the seed.
144 Vertex<int> *sv = graph.findVertex(seedId);
145 if (!sv)
146 return 1;
147
148 std::vector<int> candidates;
149 for (const auto *e : sv->getAdj())
150 candidates.push_back(e->getDest()->getInfo());
151
152 for (int c : candidates)
153 {
154 // Check that c is adjacent to all current clique members.
155 Vertex<int> *cv = graph.findVertex(c);
156 if (!cv)
157 continue;
158
159 bool adjToAll = true;
160 for (int m : clique)
161 {
162 bool found = false;
163 for (const auto *e : cv->getAdj())
164 {
165 if (e->getDest()->getInfo() == m)
166 {
167 found = true;
168 break;
169 }
170 }
171 if (!found)
172 {
173 adjToAll = false;
174 break;
175 }
176 }
177 if (adjToAll)
178 clique.push_back(c);
179 }
180
181 return static_cast<int>(clique.size());
182}
